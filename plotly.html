data(instacart)

instacart=
  instacart |>
  janitor::clean_names() |> 
  mutate(
    order_dow = factor(order_dow, 
    levels=0:6,
    labels=c("Sunday", "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday")
    ),
    order_hour_of_day=as.numeric(order_hour_of_day)
  ) |> 
  select(
    order_id, user_id, order_dow, order_hour_of_day, days_since_prior_order, product_name, aisle, department
  ) |> 
  filter(
    department %in% c("produce", "dairy eggs", "beverages"),
    !is.na(order_hour_of_day),
    !is.na(order_dow)
    )

Scatterplot: Order by Hour of day and Department

instacart |> 
  count(order_hour_of_day, department) |> 
  plot_ly(
    x= ~order_hour_of_day,
    y= ~n,
    color= ~ department,
    type = "scatter",
    mode = "lines+markers"
  ) |> 
  layout(
    title = "Order by Hour of day and Department",
    xaxis = list(title="Hour of day"),
    yaxis = list(title="No. of orders")
  )
